        // --- INITIALISATION & LOCALSTORAGE ---
        let memoryFallbackState = null;

        function updateStorageIndicator(statusText = null) {
            const el = document.getElementById('storage-status');
            if (!el) return;
            if (statusText) {
                el.textContent = statusText;
                return;
            }
            const count = Array.isArray(state.recipes) ? state.recipes.length : 0;
            el.textContent = `💾 ${count} recette(s) enregistrée(s) sur cet appareil`;
        }

        function loadState() {
            let loaded = null;
            try {
                const saved = localStorage.getItem('saveurs_budget_app_v3');
                if (saved) {
                    loaded = JSON.parse(saved);
                }
            } catch (err) {
                loaded = memoryFallbackState;
            }

            if (loaded && Array.isArray(loaded.recipes)) {
                state = { ...state, ...loaded };
                memoryFallbackState = JSON.parse(JSON.stringify(state));
            } else {
                initDefaultWeek();
            }
            updateStorageIndicator();
        }

        function initDefaultWeek() {
            state.recipes = JSON.parse(JSON.stringify(DEFAULT_RECIPES));
            // Pré-remplir 3 jours exemples + 1 goûter pour montrer le fonctionnement tout de suite
            state.mealPlan = [
                { id: 'mp-1', day: 'J1', meal: 'Soir', recipeId: 'rec-1', servings: 2 },
                { id: 'mp-2', day: 'J3', meal: 'Soir', recipeId: 'rec-5', servings: 2 },
                { id: 'mp-3', day: 'J7', meal: 'Soir', recipeId: 'rec-10', servings: 2 },
                { id: 'mp-4', day: 'J3', meal: 'Dessert / Goûter', recipeId: 'rec-11', servings: 2 }
            ];
            state.manualItems = [];
            state.checkedAutoItems = {};
            saveState();
        }

        function saveState() {
            memoryFallbackState = JSON.parse(JSON.stringify(state));
            try {
                localStorage.setItem('saveurs_budget_app_v3', JSON.stringify(state));
                updateStorageIndicator();
            } catch (err) {
                updateStorageIndicator('⚠️ Mémoire session active (pense à Exporter JSON)');
            }
            updateBadges();
        }

        // --- NAVIGATION, BADGES, MODALES & SAUVEGARDE ---
        let toastTimeout = null;

        function showToast(msg, icon = '✅') {
            const toast = document.getElementById('toast');
            const iconEl = document.getElementById('toast-icon');
            const msgEl = document.getElementById('toast-msg');
            if (!toast || !iconEl || !msgEl) return;

            iconEl.textContent = icon;
            msgEl.textContent = msg;
            toast.classList.remove('hidden');

            if (toastTimeout) clearTimeout(toastTimeout);
            toastTimeout = setTimeout(() => {
                toast.classList.add('hidden');
            }, 2600);
        }

        function closeModal(modalId) {
            const modal = document.getElementById(modalId);
            if (modal) modal.classList.add('hidden');
        }

        function switchTab(tabName) {
            state.currentTab = tabName;
            saveState();
            render();
        }

        function updateBadges() {
            const weekBadge = document.getElementById('week-badge');
            if (weekBadge) {
                const mealCount = Array.isArray(state.mealPlan) ? state.mealPlan.length : 0;
                weekBadge.textContent = mealCount;
                weekBadge.classList.toggle('hidden', mealCount === 0);
            }

            const shopBadge = document.getElementById('shop-badge');
            if (shopBadge) {
                const autoCount = typeof computeShoppingList === 'function' ? computeShoppingList().length : 0;
                const manualCount = Array.isArray(state.manualItems) ? state.manualItems.length : 0;
                const totalShop = autoCount + manualCount;
                shopBadge.textContent = totalShop;
                shopBadge.classList.toggle('hidden', totalShop === 0);
            }
        }

        function updateNavStyles() {
            const tabs = ['recipes', 'week', 'shopping'];
            tabs.forEach(tab => {
                const btn = document.getElementById(`nav-${tab}`);
                if (!btn) return;
                if (state.currentTab === tab) {
                    btn.classList.add('bg-sage-100', 'text-sage-700');
                    btn.classList.remove('text-stone-500', 'hover:bg-stone-100');
                } else {
                    btn.classList.remove('bg-sage-100', 'text-sage-700');
                    btn.classList.add('text-stone-500', 'hover:bg-stone-100');
                }
            });
        }

        function render() {
            updateNavStyles();
            updateBadges();
            const container = document.getElementById('app-content');
            if (!container) return;

            if (state.currentTab === 'recipes') {
                container.innerHTML = renderRecipesTab();
            } else if (state.currentTab === 'shopping') {
                container.innerHTML = renderShoppingTab();
            } else {
                container.innerHTML = renderWeekTab();
            }
        }

            initDefaultWeek();

        function exportBackupData() {
            try {
                const dataToExport = {
                    version: 'v3',
                    exportedAt: new Date().toISOString(),
                    recipes: state.recipes,
                    mealPlan: state.mealPlan,
                    manualItems: state.manualItems,
                    checkedAutoItems: state.checkedAutoItems,
                    defaultServings: state.defaultServings
                };
                const blob = new Blob([JSON.stringify(dataToExport, null, 2)], { type: 'application/json' });
                const url = URL.createObjectURL(blob);
                const a = document.createElement('a');
                const dateStr = new Date().toISOString().slice(0, 10);
                a.href = url;
                a.download = `saveurs-budget-sauvegarde-${dateStr}.json`;
                document.body.appendChild(a);
                a.click();
                document.body.removeChild(a);
                URL.revokeObjectURL(url);
                showToast('Sauvegarde JSON téléchargée !', '💾');
            } catch (err) {
                showToast('Erreur lors de l’export JSON', '⚠️');
            }
        }

        function importBackupData(input) {
            const file = input.files && input.files[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = function(e) {
                try {
                    const parsed = JSON.parse(e.target.result);
                    if (!parsed || !Array.isArray(parsed.recipes)) {
                        showToast('Fichier de sauvegarde invalide', '⚠️');
                        return;
                    }
                    state.recipes = parsed.recipes;
                    state.mealPlan = Array.isArray(parsed.mealPlan) ? parsed.mealPlan : [];
                    state.manualItems = Array.isArray(parsed.manualItems) ? parsed.manualItems : [];
                    state.checkedAutoItems = parsed.checkedAutoItems && typeof parsed.checkedAutoItems === 'object' ? parsed.checkedAutoItems : {};
                    if (typeof parsed.defaultServings === 'number' && parsed.defaultServings >= 1) {
                        state.defaultServings = parsed.defaultServings;
                    }
                    saveState();
                    render();
                    showToast('Sauvegarde restaurée avec succès !', '📂');
                } catch (err) {
                    showToast('Impossible de lire le fichier JSON', '⚠️');
                } finally {
                    input.value = '';
                }
            };
            reader.readAsText(file);
        }

        function fileToBase64(file) {
            return new Promise((resolve, reject) => {
                const reader = new FileReader();
                reader.onload = () => resolve(reader.result.split(',')[1]);
                reader.onerror = reject;
                reader.readAsDataURL(file);
            });
        }

        function arrayBufferToBase64(buffer) {
            let binary = '';
            const bytes = new Uint8Array(buffer);
            for (let i = 0; i < bytes.byteLength; i++) {
                binary += String.fromCharCode(bytes[i]);
            }
            return btoa(binary);
        }

        function encodeMonoWav(float32Samples, sampleRate) {
            const buffer = new ArrayBuffer(44 + float32Samples.length * 2);
            const view = new DataView(buffer);
            const writeStr = (offset, str) => {
                for (let i = 0; i < str.length; i++) view.setUint8(offset + i, str.charCodeAt(i));
            };
            writeStr(0, 'RIFF');
            view.setUint32(4, 36 + float32Samples.length * 2, true);
            writeStr(8, 'WAVE');
            writeStr(12, 'fmt ');
            view.setUint32(16, 16, true);
            view.setUint16(20, 1, true);
            view.setUint16(22, 1, true);
            view.setUint32(24, sampleRate, true);
            view.setUint32(28, sampleRate * 2, true);
            view.setUint16(32, 2, true);
            view.setUint16(34, 16, true);
            writeStr(36, 'data');
            view.setUint32(40, float32Samples.length * 2, true);

            let offset = 44;
            for (let i = 0; i < float32Samples.length; i++, offset += 2) {
                const s = Math.max(-1, Math.min(1, float32Samples[i]));
                view.setInt16(offset, s < 0 ? s * 0x8000 : s * 0x7FFF, true);
            }
            return buffer;
        }

        // Compresse une vidéo > 18 Mo en extrayant sa piste audio (8kHz) + 8 captures d'écran clés
        async function compressLargeVideo(file) {
            const mediaParts = [];

            // 1. Extraction et ré-échantillonnage de la piste audio en WAV 8kHz mono (très léger : ~900 Ko/min)
            try {
                const arrayBuffer = await file.arrayBuffer();
                const AudioCtx = window.AudioContext || window.webkitAudioContext;
                const audioCtx = new AudioCtx();
                const decoded = await audioCtx.decodeAudioData(arrayBuffer);
                const targetRate = 8000;
                const maxSeconds = Math.min(decoded.duration, 180); // Max 3 min
                const frameCount = Math.floor(maxSeconds * targetRate);
                const offlineCtx = new OfflineAudioContext(1, frameCount, targetRate);
                const source = offlineCtx.createBufferSource();
                source.buffer = decoded;
                source.connect(offlineCtx.destination);
                source.start();
                const rendered = await offlineCtx.startRendering();
                const wavBuffer = encodeMonoWav(rendered.getChannelData(0), targetRate);
                mediaParts.push({
                    inlineData: {
                        mimeType: 'audio/wav',
                        data: arrayBufferToBase64(wavBuffer)
                    }
                });
                audioCtx.close();
            } catch (e) {
                // Si la vidéo n'a pas de piste audio, on continue avec les images
            }

            // 2. Extraction de 8 images clés réparties sur la vidéo pour lire les textes incrustés
            const videoUrl = URL.createObjectURL(file);
            try {
                const video = document.createElement('video');
                video.src = videoUrl;
                video.muted = true;
                video.playsInline = true;
                await new Promise((res, rej) => {
                    video.onloadedmetadata = res;
                    video.onerror = rej;
                });

                const canvas = document.createElement('canvas');
                const scale = Math.min(1, 512 / Math.max(video.videoWidth || 512, video.videoHeight || 512));
                canvas.width = Math.round((video.videoWidth || 360) * scale);
                canvas.height = Math.round((video.videoHeight || 640) * scale);
                const ctx = canvas.getContext('2d');

                const duration = video.duration || 10;
                const frameCount = 8;
                for (let i = 0; i < frameCount; i++) {
                    const targetTime = (duration * (i + 0.5)) / frameCount;
                    video.currentTime = targetTime;
                    await new Promise(res => { video.onseeked = res; });
                    ctx.drawImage(video, 0, 0, canvas.width, canvas.height);
                    const jpegDataUrl = canvas.toDataURL('image/jpeg', 0.65);
                    mediaParts.push({
                        inlineData: {
                            mimeType: 'image/jpeg',
                            data: jpegDataUrl.split(',')[1]
                        }
                    });
                }
            } finally {
                URL.revokeObjectURL(videoUrl);
            }

            return mediaParts;
        }

        async function extractRecipeWithAI() {
            const rawText = document.getElementById('social-raw-text').value.trim();
            const fileInput = document.getElementById('social-video-file');
            const videoFile = fileInput.files && fileInput.files[0];

            if (!rawText && !videoFile) {

            }

            const btn = document.getElementById('ai-extract-btn');

            btn.disabled = true;
            btnIcon.textContent = '⏳';
            btnText.textContent = videoFile ? 'Analyse de la vidéo en cours...' : 'Extraction IA en cours...';

            try {
                const parts = [
                    {
                        text: `Analyse ce contenu issu d'une vidéo culinaire (TikTok / Instagram) et extrais-en une fiche recette structurée, claire et économique en français.
Classe la recette dans l'un de ces thèmes exacts : "Viande", "Italien", "Pâtes", "Monde", "Poisson", "Mijoté", "Fast food", "Desserts & Goûters".
Estime également le nombre de points Weight Watchers (programme actuel) PAR PORTION (entier réaliste) en tenant compte des aliments à zéro point (fruits, légumes, œufs, blanc de poulet, poisson blanc, yaourt nature 0%).
Classe chaque ingrédient dans l'un de ces rayons exacts : "Fruits & Légumes", "Épicerie & Pâtes", "Crèmerie", "Boucherie & Poissonnerie".
Utilise uniquement ces unités : "g", "ml", "cl", "pièce", "gousses".
${rawText ? `\nTexte / légende de la vidéo :\n${rawText}` : ''}`
                    }
                ];

                if (videoFile) {
                    if (videoFile.size <= 18 * 1024 * 1024) {
                        const base64Video = await fileToBase64(videoFile);
                        parts.push({
                            inlineData: {
                                mimeType: videoFile.type || 'video/mp4',
                                data: base64Video
                            }
                        });
                    } else {
                        btnText.textContent = 'Compression de la vidéo (audio + images)...';
                        const compressedParts = await compressLargeVideo(videoFile);
                        parts.push(...compressedParts);
                        btnText.textContent = 'Extraction IA en cours...';
                    }
                }

                const payload = {
                    contents: [{ role: 'user', parts }],
                    generationConfig: {
                        responseMimeType: "application/json",
                        responseSchema: {
                            type: "OBJECT",
                            properties: {
                                title: { type: "STRING" },
                                theme: { type: "STRING", enum: ["Viande", "Italien", "Pâtes", "Monde", "Poisson", "Mijoté", "Fast food", "Desserts & Goûters"] },
                                servings: { type: "NUMBER" },
                                time: { type: "NUMBER" },
                                estimatedCostEuro: { type: "NUMBER" },
                                wwPointsPerServing: { type: "NUMBER" },
                                steps: { type: "STRING" },
                                ingredients: {
                                    type: "ARRAY",
                                    items: {
                                        type: "OBJECT",
                                        properties: {
                                            name: { type: "STRING" },
                                            qty: { type: "NUMBER" },
                                            unit: { type: "STRING", enum: ["g", "ml", "cl", "pièce", "gousses"] },
                                            aisle: { type: "STRING", enum: ["Fruits & Légumes", "Épicerie & Pâtes", "Crèmerie", "Boucherie & Poissonnerie"] }
                                        },
                                        required: ["name", "qty", "unit", "aisle"]
                                    }
                                }
                            },
                            required: ["title", "theme", "servings", "time", "estimatedCostEuro", "wwPointsPerServing", "steps", "ingredients"]
                        }
                    }
                };

                const apiKey = "";
                const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=${apiKey}`;

                // Appel avec backoff exponentiel
                let response;
                const delays = [1000, 2000, 4000];
                for (let attempt = 0; attempt <= delays.length; attempt++) {
                    response = await fetch(apiUrl, {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify(payload)
                    });
                    if (response.ok || attempt === delays.length) break;
                    await new Promise(r => setTimeout(r, delays[attempt]));
                }

                if (!response || !response.ok) {
                    throw new Error('Erreur réseau lors de l’analyse IA');
                }

                const result = await response.json();
                const jsonText = result?.candidates?.[0]?.content?.parts?.[0]?.text;

                const extracted = JSON.parse(jsonText);

                    extracted.ingredients.forEach(ing => {
                        addIngredientRow(ing.name, ing.qty, ing.unit, ing.aisle);
                    });
                } else {
                    addIngredientRow();
                }

                showToast('Recette extraite et pré-remplie ! Vérifie puis enregistre.', '✨');
            } catch (err) {
                showToast('Impossible d’extraire la recette. Vérifie le texte ou la taille de la vidéo.', '⚠️');
            } finally {
                btn.disabled = false;
                btnIcon.textContent = '🪄';
                btnText.textContent = 'Extraire la recette automatiquement';
            }
        }

        function autoFillWeek() {
            const newPlan = [];
            WEEK_SLOTS.forEach((slot, idx) => {
                // Cherche les recettes correspondant au thème du jour
                let candidates = state.recipes.filter(r => r.theme === slot.theme);
                if (candidates.length === 0) candidates = state.recipes.filter(r => r.theme !== 'Desserts & Goûters');
                if (candidates.length > 0) {
                    const randomRec = candidates[Math.floor(Math.random() * candidates.length)];
                    newPlan.push({
                        id: 'mp-auto-' + idx + '-' + Date.now(),
                        day: slot.code,
                        meal: 'Soir',
                        recipeId: randomRec.id,
                        servings: state.defaultServings
                    });
                }
            });
            // Ajoute également 2 idées desserts/goûters sur la semaine si disponibles
            const dessertCandidates = state.recipes.filter(r => r.theme === 'Desserts & Goûters');
            if (dessertCandidates.length > 0) {
                const d1 = dessertCandidates[0];
                newPlan.push({
                    id: 'mp-auto-des1-' + Date.now(),
                    day: 'J3',
                    meal: 'Dessert / Goûter',
                    recipeId: d1.id,
                    servings: state.defaultServings
                });
                if (dessertCandidates.length > 1) {
                    const d2 = dessertCandidates[1];
                    newPlan.push({
                        id: 'mp-auto-des2-' + Date.now(),
                        day: 'J6',
                        meal: 'Dessert / Goûter',
                        recipeId: d2.id,
                        servings: state.defaultServings
                    });
                }
            }
            state.mealPlan = newPlan;
            saveState();
            render();
            showToast('Semaine J1 à J7 + goûters générés !', '✨');
        }

        function quickPlanForSlot(dayCode, theme) {
            if (state.recipes.length === 0) {
                showToast('Ajoute d\'abord une recette !', '⚠️');
                return;
            }
            // Mettre en avant les recettes du thème correspondant, puis les desserts/goûters, puis les autres
            const themeRecipes = state.recipes.filter(r => r.theme === theme);
            const dessertRecipes = state.recipes.filter(r => r.theme === 'Desserts & Goûters' && r.theme !== theme);
            const otherRecipes = state.recipes.filter(r => r.theme !== theme && r.theme !== 'Desserts & Goûters');
            const orderedRecipes = [...themeRecipes, ...dessertRecipes, ...otherRecipes];

            document.getElementById('plan-day').value = dayCode;
            openPlanModal(orderedRecipes[0].id);

            let optionsHtml = '';
            if (themeRecipes.length > 0) {
                optionsHtml += `<optgroup label="✨ Idées ${theme}">` +
                    themeRecipes.map(r => `<option value="${r.id}">${r.emoji || ''} ${r.title} (${r.wwPoints ?? '?'} pts)</option>`).join('') +
                    `</optgroup>`;
            }
            if (dessertRecipes.length > 0) {
                optionsHtml += `<optgroup label="🧁 Desserts & Goûters">` +
                    dessertRecipes.map(r => `<option value="${r.id}">${r.emoji || ''} ${r.title} (${r.wwPoints ?? '?'} pts)</option>`).join('') +
                    `</optgroup>`;
            }
            if (otherRecipes.length > 0) {
                optionsHtml += `<optgroup label="Autres recettes">` +
                    otherRecipes.map(r => `<option value="${r.id}">${r.emoji || ''} ${r.title} (${r.wwPoints ?? '?'} pts)</option>`).join('') +
                    `</optgroup>`;
            }

            document.getElementById('plan-recipe-name').innerHTML = `
                <select onchange="handleQuickRecipeChange(this.value)" class="w-full bg-transparent font-bold text-sage-800 focus:outline-none">
                    ${optionsHtml}
                </select>
            `;
        }

        function handleQuickRecipeChange(recipeId) {
            state.selectedRecipeForPlan = recipeId;
            const rec = state.recipes.find(r => r.id === recipeId);
            if (rec && rec.theme === 'Desserts & Goûters') {
                document.getElementById('plan-meal').value = 'Dessert / Goûter';
            } else {
                document.getElementById('plan-meal').value = 'Soir';
            }
        }

        function computeShoppingList() {
            const map = new Map();

            state.mealPlan.forEach(planItem => {
                const rec = state.recipes.find(r => r.id === planItem.recipeId);
                if (!rec) return;

                const ratio = planItem.servings / rec.servings;

                rec.ingredients.forEach(ing => {
                    // Normalisation du nom pour regrouper les mêmes ingrédients (ex: Oignon jaune)
                    const cleanName = ing.name.trim();
                    const key = `${cleanName.toLowerCase()}_${ing.unit.toLowerCase()}`;
                    const scaledQty = ing.qty * ratio;

                    if (map.has(key)) {
                        const existing = map.get(key);
                        existing.qty += scaledQty;
                    } else {
                        map.set(key, {
                            key,
                            name: cleanName,
                            qty: scaledQty,
                            unit: ing.unit,
                            aisle: ing.aisle || 'Épicerie'
                        });
                    }
                });
            });

            return Array.from(map.values());
        }

        function toggleAutoItem(key) {
            state.checkedAutoItems[key] = !state.checkedAutoItems[key];
            saveState();
            render();
        }

        function addManualShoppingItem(e) {
            e.preventDefault();
            const input = document.getElementById('manual-item-input');
            const aisle = document.getElementById('manual-item-aisle').value;
            const name = input.value.trim();
            if (!name) return;

            state.manualItems.push({
                id: 'man-' + Date.now(),
                name,
                aisle,
                checked: false
            });
            input.value = '';
            saveState();
            render();
            showToast('Article ajouté à la liste', '🛒');
        }

        function toggleManualItem(id) {
            const item = state.manualItems.find(i => i.id === id);
            if (item) {
                item.checked = !item.checked;
                saveState();
                render();
            }
        }

        function removeManualItem(id) {
            state.manualItems = state.manualItems.filter(i => i.id !== id);
            saveState();
            render();
        }

        function formatQty(num) {
            return Number.isInteger(num) ? num : Math.round(num * 10) / 10;
        }

        function renderShoppingTab() {
            const autoItems = computeShoppingList();
            const aislesOrder = ['Fruits & Légumes', 'Boucherie & Poissonnerie', 'Crèmerie', 'Épicerie & Pâtes', 'Autres / Maison'];
            const aisleIcons = {
                'Fruits & Légumes': '🥦',
                'Boucherie & Poissonnerie': '🥩',
                'Crèmerie': '🧀',
                'Épicerie & Pâtes': '🍝',
                'Autres / Maison': '🧻'
            };

            // Regroupement par rayon
            const grouped = {};
            aislesOrder.forEach(a => grouped[a] = []);

            autoItems.forEach(item => {
                const targetAisle = grouped[item.aisle] ? item.aisle : 'Épicerie & Pâtes';
                grouped[targetAisle].push({
                    type: 'auto',
                    id: item.key,
                    name: item.name,
                    qtyText: `${formatQty(item.qty)} ${item.unit}`,
                    checked: !!state.checkedAutoItems[item.key]
                });
            });

            state.manualItems.forEach(item => {
                const targetAisle = grouped[item.aisle] ? item.aisle : 'Autres / Maison';
                grouped[targetAisle].push({
                    type: 'manual',
                    id: item.id,
                    name: item.name,
                    qtyText: 'Ajout manuel',
                    checked: item.checked
                });
            });

            let aislesHtml = '';
            let hasAnyItem = false;

            aislesOrder.forEach(aisle => {
                const items = grouped[aisle];
                if (items.length === 0) return;
                hasAnyItem = true;

                const itemsHtml = items.map(item => `
                    <div class="flex items-center justify-between py-2.5 border-b border-stone-100 last:border-0">
                        <label class="flex items-center gap-3 cursor-pointer flex-1 select-none">
                            <input type="checkbox" ${item.checked ? 'checked' : ''} 
                                onchange="${item.type === 'auto' ? `toggleAutoItem('${item.id}')` : `toggleManualItem('${item.id}')`}"
                                class="w-5 h-5 rounded-lg accent-sage-600 cursor-pointer">
                            <span class="text-sm font-medium transition ${item.checked ? 'line-through text-stone-400' : 'text-stone-800'}">
                                ${item.name}
                            </span>
                        </label>
                        <div class="flex items-center gap-2">
                            <span class="text-xs font-semibold px-2.5 py-1 rounded-lg ${item.checked ? 'bg-stone-100 text-stone-400' : 'bg-sage-50 text-sage-700'}">
                                ${item.qtyText}
                            </span>
                            ${item.type === 'manual' ? `<button onclick="removeManualItem('${item.id}')" class="text-stone-400 hover:text-red-500 text-xs px-1">✕</button>` : ''}
                        </div>
                    </div>
                `).join('');

                aislesHtml += `
                    <div class="bg-white rounded-3xl p-4 shadow-xs border border-stone-200/70 animate-fade">
                        <h3 class="font-display font-bold text-stone-900 text-sm mb-2 flex items-center gap-2">
                            <span>${aisleIcons[aisle] || '🛒'}</span>
                            <span>${aisle}</span>
                            <span class="text-xs font-normal text-stone-400">(${items.length})</span>
                        </h3>
                        <div>${itemsHtml}</div>
                    </div>
                `;
            });

            if (!hasAnyItem) {
                aislesHtml = `
                    <div class="bg-white rounded-3xl p-8 text-center border border-stone-200/70 space-y-2">
                        <div class="text-4xl">🛒</div>
                        <h3 class="font-display font-bold text-stone-800">Ta liste est vide</h3>
                        <p class="text-xs text-stone-500">Planifie des recettes dans l'onglet "Ma Semaine" pour générer automatiquement tes courses.</p>
                    </div>
                `;
            }

            return `
                <div class="flex items-center justify-between">
                    <div>
                        <h2 class="font-display text-xl font-bold text-stone-900">Liste de Courses</h2>
                        <p class="text-xs text-stone-500">Quantités cumulées selon ton planning</p>
                    </div>
                    <button onclick="copyShoppingList()" class="py-2 px-3.5 bg-sage-600 hover:bg-sage-700 text-white text-xs font-bold rounded-2xl shadow-xs transition flex items-center gap-1.5">
                        <span>📋 Copier</span>
                    </button>
                </div>

                <!-- Formulaire d'ajout manuel rapide -->
                <form onsubmit="addManualShoppingItem(event)" class="bg-sand p-3 rounded-2xl flex gap-2 items-center border border-stone-200/80">
                    <input type="text" id="manual-item-input" placeholder="Ajouter un article (ex: Café, Sopalin...)" class="flex-1 min-w-0 px-3 py-2 bg-white rounded-xl text-xs border border-stone-200 focus:outline-none focus:border-sage-500">
                    <select id="manual-item-aisle" class="px-2 py-2 bg-white rounded-xl text-xs border border-stone-200">
                        <option value="Autres / Maison">Maison</option>
                        <option value="Fruits & Légumes">Fruits/Lég.</option>
                        <option value="Crèmerie">Crèmerie</option>
                        <option value="Épicerie & Pâtes">Épicerie</option>
                    </select>
                    <button type="submit" class="px-3 py-2 bg-stone-800 text-white text-xs font-bold rounded-xl hover:bg-stone-900 transition">
                        +
                    </button>
                </form>

                <div class="space-y-3">
                    ${aislesHtml}
                </div>
            `;
        }

        function copyShoppingList() {
            const autoItems = computeShoppingList();
            if (autoItems.length === 0 && state.manualItems.length === 0) {
                showToast('Rien à copier pour le moment', '⚠️');
                return;
            }

            let text = "🛒 MA LISTE DE COURSES :\n\n";
            autoItems.forEach(item => {
                const check = state.checkedAutoItems[item.key] ? '☑' : '☐';
                text += `${check} ${item.name} (${formatQty(item.qty)} ${item.unit})\n`;
            });
            state.manualItems.forEach(item => {
                const check = item.checked ? '☑' : '☐';
                text += `${check} ${item.name}\n`;
            });

            // Copie compatible iframe via textarea temporaire
            const textarea = document.createElement('textarea');
            textarea.value = text;
            document.body.appendChild(textarea);
            textarea.select();
            document.execCommand('copy');
            document.body.removeChild(textarea);

            showToast('Liste copiée ! Prête à coller dans Notes/WhatsApp', '📋');
        }
